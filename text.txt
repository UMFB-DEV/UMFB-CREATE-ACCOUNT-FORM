// BankUseRequirementChecklist.jsx
import React, { useState } from 'react';

const BankUseRequirementChecklist = () => {
  const [checklist, setChecklist] = useState({
    // Document Requirements
    documents: [
      { id: 1, name: 'Account opening form duly completed', checked: false, deferred: false, waived: false },
      { id: 2, name: 'Specimen signature card duly completed', checked: false, deferred: false, waived: false },
      { id: 3, name: 'Copy of CAC Certificate of Registration', checked: false, deferred: false, waived: false },
      { id: 4, name: 'Board Resolution', checked: false, deferred: false, waived: false },
      { id: 5, name: 'Copy of Memorandum and Article of Association (Certified as true copy by the Registrar of Companies)', checked: false, deferred: false, waived: false },
      { id: 6, name: 'Form C07 - Particulars of Directors (Certified as true copy by the Registrar of Companies and Certification by a Notary public for Foreign Companies)', checked: false, deferred: false, waived: false },
      { id: 7, name: 'Form C02 - Allotment of Shares (Certified as true copy by the Registrar of Companies and Certification by a Notary public for Foreign Companies)', checked: false, deferred: false, waived: false },
      { id: 8, name: 'Partnership Deed (where applicable)', checked: false, deferred: false, waived: false },
      { id: 9, name: 'Approval Letter (for Government Agency)', checked: false, deferred: false, waived: false },
      { id: 10, name: 'Act/Gazette (for Government Agency where applicable)', checked: false, deferred: false, waived: false },
      { id: 11, name: 'Two (2) passport sized photographs of each signatory to the account with name written on the reverse side', checked: false, deferred: false, waived: false },
      { id: 12, name: 'Residence permit (for non-Nigerians)', checked: false, deferred: false, waived: false },
      { id: 13, name: 'Evidence of Registration with Nigerian Investment Promotion Council (NIPC) (where applicable)', checked: false, deferred: false, waived: false },
      { id: 14, name: 'Evidence of Registration with Special Control Unit against Money Laundering (SCUML) (where applicable)', checked: false, deferred: false, waived: false },
      { id: 15, name: 'Search Report', checked: false, deferred: false, waived: false },
      { id: 16, name: 'Power of Attorney (where applicable)', checked: false, deferred: false, waived: false },
      { id: 17, name: 'Proof of Company address', checked: false, deferred: false, waived: false },
      { id: 18, name: 'Business Premises Visitation certificate', checked: false, deferred: false, waived: false },
      { id: 19, name: 'Proof of Identity of all Signatories and Directors/Officers whose names appear on the account opening form/document (Preferred Identity card are International Passport, National Identity Card, National Driver\'s License and Valid Nigerian INEC Voter\'s Card)', checked: false, deferred: false, waived: false },
      { id: 20, name: 'Proof of Address of all Signatories and Directors/Officers whose names appear on the account opening form/document Utility bill (Certified true copy is acceptable if original is held)', checked: false, deferred: false, waived: false },
      { id: 21, name: 'Two (2) completed satisfactory Reference forms', checked: false, deferred: false, waived: false },
      { id: 22, name: 'Others (please specify)', checked: false, deferred: false, waived: false, otherSpecify: '' },
    ],
    
    // KYC Watch-list Compliance
    kycWatchlist: {
      isOnWatchlist: false,
      csoName: '',
      csoSignature: '',
      csoTextSignature: '',
      csoDate: '',
    },
    
    // Approvals
    approvals: {
      customerServiceOfficer: {
        name: '',
        signature: '',
        textSignature: '',
        date: '',
      },
      headOfOperations: {
        name: '',
        signature: '',
        textSignature: '',
        date: '',
      },
      branchHead: {
        name: '',
        signature: '',
        textSignature: '',
        date: '',
      },
      groupZonalHead: {
        name: '',
        signature: '',
        textSignature: '',
        date: '',
      },
    },
    
    // Comments
    comments: '',
  });

  const handleDocumentChange = (id, field, value) => {
    setChecklist(prev => {
      const newDocuments = prev.documents.map(doc => {
        if (doc.id === id) {
          if (field === 'checked' && value) {
            return { ...doc, checked: true, deferred: false, waived: false };
          }
          if (field === 'deferred' && value) {
            return { ...doc, checked: false, deferred: true, waived: false };
          }
          if (field === 'waived' && value) {
            return { ...doc, checked: false, deferred: false, waived: true };
          }
          return { ...doc, [field]: value };
        }
        return doc;
      });
      return { ...prev, documents: newDocuments };
    });
  };

  const handleOtherSpecifyChange = (id, value) => {
    setChecklist(prev => {
      const newDocuments = prev.documents.map(doc => {
        if (doc.id === id) {
          return { ...doc, otherSpecify: value };
        }
        return doc;
      });
      return { ...prev, documents: newDocuments };
    });
  };

  const handleKycChange = (field, value) => {
    setChecklist(prev => ({
      ...prev,
      kycWatchlist: { ...prev.kycWatchlist, [field]: value }
    }));
  };

  const handleApprovalChange = (role, field, value) => {
    setChecklist(prev => ({
      ...prev,
      approvals: {
        ...prev.approvals,
        [role]: { ...prev.approvals[role], [field]: value }
      }
    }));
  };

  const handleCommentsChange = (e) => {
    setChecklist(prev => ({ ...prev, comments: e.target.value }));
  };

  const getStatusBadge = (doc) => {
    if (doc.checked) return { text: '✓ Checked', class: 'bg-green-100 text-green-800' };
    if (doc.deferred) return { text: '⏳ Deferred', class: 'bg-yellow-100 text-yellow-800' };
    if (doc.waived) return { text: '⭕ Waived', class: 'bg-blue-100 text-blue-800' };
    return { text: '✗ Pending', class: 'bg-red-100 text-red-800' };
  };

  const getRowClass = (doc) => {
    if (doc.checked) return 'bg-green-50';
    if (doc.deferred) return 'bg-yellow-50';
    if (doc.waived) return 'bg-blue-50';
    return 'bg-red-50';
  };

  const handleSubmit = (e) => {
    e.preventDefault();
    console.log('Checklist submitted:', checklist);
    alert('Bank Use Checklist submitted successfully!');
  };

  // Statistics
  const totalDocs = checklist.documents.length;
  const checkedCount = checklist.documents.filter(d => d.checked).length;
  const deferredCount = checklist.documents.filter(d => d.deferred).length;
  const waivedCount = checklist.documents.filter(d => d.waived).length;
  const pendingCount = totalDocs - checkedCount - deferredCount - waivedCount;

  return (
    <div className="max-w-7xl mx-auto p-4 sm:p-6 font-sans">
      {/* Header */}
      <div className="text-center border-b-4 border-[#003366] pb-5 mb-8">
        <h1 className="text-2xl sm:text-3xl md:text-4xl font-bold text-[#003366] uppercase tracking-wide">
          BANK USE REQUIREMENT CHECKLIST
        </h1>
        <p className="text-gray-600 text-sm sm:text-base mt-1">
          Document Verification and Compliance Check
        </p>
      </div>

      {/* Statistics Summary */}
      <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-5 gap-3 md:gap-4 mb-8">
        <div className="text-center p-3 md:p-4 rounded-lg bg-gray-50 border-2 border-[#003366]">
          <span className="block text-2xl md:text-3xl font-bold text-[#003366]">{totalDocs}</span>
          <span className="text-xs md:text-sm text-gray-600">Total Documents</span>
        </div>
        <div className="text-center p-3 md:p-4 rounded-lg bg-gray-50 border-2 border-green-500">
          <span className="block text-2xl md:text-3xl font-bold text-green-600">{checkedCount}</span>
          <span className="text-xs md:text-sm text-gray-600">✓ Checked</span>
        </div>
        <div className="text-center p-3 md:p-4 rounded-lg bg-gray-50 border-2 border-yellow-500">
          <span className="block text-2xl md:text-3xl font-bold text-yellow-600">{deferredCount}</span>
          <span className="text-xs md:text-sm text-gray-600">⏳ Deferred</span>
        </div>
        <div className="text-center p-3 md:p-4 rounded-lg bg-gray-50 border-2 border-blue-500">
          <span className="block text-2xl md:text-3xl font-bold text-blue-600">{waivedCount}</span>
          <span className="text-xs md:text-sm text-gray-600">⭕ Waived</span>
        </div>
        <div className="text-center p-3 md:p-4 rounded-lg bg-gray-50 border-2 border-red-500">
          <span className="block text-2xl md:text-3xl font-bold text-red-600">{pendingCount}</span>
          <span className="text-xs md:text-sm text-gray-600">✗ Pending</span>
        </div>
      </div>

      <form onSubmit={handleSubmit}>
        {/* Document Checklist Table */}
        <div className="bg-white p-4 md:p-6 rounded-lg shadow-md mb-6 overflow-hidden">
          <h2 className="text-lg md:text-xl font-bold text-[#003366] border-b-2 border-[#003366] pb-2 mb-4">
            DOCUMENTS REQUIRED
          </h2>
          <div className="overflow-x-auto">
            <table className="w-full border-collapse text-xs md:text-sm">
              <thead>
                <tr className="bg-[#003366] text-white">
                  <th className="px-2 md:px-4 py-2 md:py-3 text-left font-semibold">S/N</th>
                  <th className="px-2 md:px-4 py-2 md:py-3 text-left font-semibold">DOCUMENTS REQUIRED</th>
                  <th className="px-2 md:px-4 py-2 md:py-3 text-center font-semibold">CHECKED</th>
                  <th className="px-2 md:px-4 py-2 md:py-3 text-center font-semibold">DEFERRED</th>
                  <th className="px-2 md:px-4 py-2 md:py-3 text-center font-semibold">WAIVED</th>
                  <th className="px-2 md:px-4 py-2 md:py-3 text-center font-semibold">STATUS</th>
                </tr>
              </thead>
              <tbody>
                {checklist.documents.map((doc, index) => {
                  const status = getStatusBadge(doc);
                  const rowClass = getRowClass(doc);
                  return (
                    <tr key={doc.id} className={`${rowClass} border-b border-gray-200 hover:bg-gray-100 transition-colors`}>
                      <td className="px-2 md:px-4 py-2 md:py-3 text-center">{index + 1}</td>
                      <td className="px-2 md:px-4 py-2 md:py-3 min-w-[200px] md:min-w-[300px]">
                        <span className="block">{doc.name}</span>
                        {doc.id === 22 && (
                          <input
                            type="text"
                            className="mt-1 w-full px-3 py-1 border border-gray-300 rounded focus:outline-none focus:ring-2 focus:ring-[#003366] focus:border-transparent text-sm"
                            value={doc.otherSpecify || ''}
                            onChange={(e) => handleOtherSpecifyChange(doc.id, e.target.value)}
                            placeholder="Please specify..."
                          />
                        )}
                      </td>
                      <td className="px-2 md:px-4 py-2 md:py-3 text-center">
                        <input
                          type="radio"
                          name={`doc_${doc.id}`}
                          checked={doc.checked}
                          onChange={() => handleDocumentChange(doc.id, 'checked', true)}
                          className="w-4 h-4 md:w-5 md:h-5 cursor-pointer accent-green-600"
                        />
                      </td>
                      <td className="px-2 md:px-4 py-2 md:py-3 text-center">
                        <input
                          type="radio"
                          name={`doc_${doc.id}`}
                          checked={doc.deferred}
                          onChange={() => handleDocumentChange(doc.id, 'deferred', true)}
                          className="w-4 h-4 md:w-5 md:h-5 cursor-pointer accent-yellow-500"
                        />
                      </td>
                      <td className="px-2 md:px-4 py-2 md:py-3 text-center">
                        <input
                          type="radio"
                          name={`doc_${doc.id}`}
                          checked={doc.waived}
                          onChange={() => handleDocumentChange(doc.id, 'waived', true)}
                          className="w-4 h-4 md:w-5 md:h-5 cursor-pointer accent-blue-500"
                        />
                      </td>
                      <td className="px-2 md:px-4 py-2 md:py-3 text-center">
                        <span className={`inline-block px-2 py-1 rounded-full text-xs font-semibold whitespace-nowrap ${status.class}`}>
                          {status.text}
                        </span>
                      </td>
                    </tr>
                  );
                })}
              </tbody>
            </table>
          </div>
        </div>

        {/* Comments Section */}
        <div className="bg-white p-4 md:p-6 rounded-lg shadow-md mb-6">
          <h2 className="text-lg md:text-xl font-bold text-[#003366] mb-3">COMMENTS</h2>
          <textarea
            value={checklist.comments}
            onChange={handleCommentsChange}
            placeholder="Enter any comments or observations regarding the document verification..."
            rows="4"
            className="w-full px-4 py-3 border-2 border-gray-300 rounded-lg focus:outline-none focus:ring-2 focus:ring-[#003366] focus:border-transparent resize-y text-sm md:text-base"
          />
        </div>

        {/* KYC Watch-list Compliance */}
        <div className="bg-white p-4 md:p-6 rounded-lg shadow-md mb-6">
          <h2 className="text-lg md:text-xl font-bold text-[#003366] border-b-2 border-[#003366] pb-2 mb-4">
            G. AUTHENTICATION FOR KYC WATCH-LIST COMPLIANCE
          </h2>
          
          <div className="mb-5">
            <label className="block font-semibold text-base md:text-lg mb-2 text-gray-800">
              Is the applicant on the KYC Watch-list?
            </label>
            <div className="flex gap-4 md:gap-8">
              <label className="flex items-center gap-2 text-sm md:text-base cursor-pointer">
                <input
                  type="radio"
                  name="kycWatchlist"
                  value="yes"
                  checked={checklist.kycWatchlist.isOnWatchlist === true}
                  onChange={() => handleKycChange('isOnWatchlist', true)}
                  className="w-4 h-4 md:w-5 md:h-5 cursor-pointer"
                />
                Yes
              </label>
              <label className="flex items-center gap-2 text-sm md:text-base cursor-pointer">
                <input
                  type="radio"
                  name="kycWatchlist"
                  value="no"
                  checked={checklist.kycWatchlist.isOnWatchlist === false}
                  onChange={() => handleKycChange('isOnWatchlist', false)}
                  className="w-4 h-4 md:w-5 md:h-5 cursor-pointer"
                />
                No
              </label>
            </div>
          </div>

          <div className="grid grid-cols-1 md:grid-cols-2 gap-4 md:gap-6 pt-4 border-t border-gray-200">
            <div>
              <label className="block font-semibold text-sm mb-1 text-gray-700">
                Name and Signature of CSO <span className="text-red-500">*</span>
              </label>
              <input
                type="text"
                value={checklist.kycWatchlist.csoName}
                onChange={(e) => handleKycChange('csoName', e.target.value)}
                placeholder="Enter CSO name"
                className="w-full px-3 py-2 border border-gray-300 rounded focus:outline-none focus:ring-2 focus:ring-[#003366] focus:border-transparent text-sm"
                required
              />
            </div>

            <div>
              <label className="block font-semibold text-sm mb-1 text-gray-700">
                CSO Signature (Type name) <span className="text-red-500">*</span>
              </label>
              <input
                type="text"
                value={checklist.kycWatchlist.csoTextSignature}
                onChange={(e) => handleKycChange('csoTextSignature', e.target.value)}
                placeholder="Type CSO signature"
                className="w-full px-3 py-2 border border-gray-300 rounded focus:outline-none focus:ring-2 focus:ring-[#003366] focus:border-transparent text-sm font-cursive"
                style={{ fontFamily: "'Segoe Script', 'Brush Script MT', cursive" }}
                required
              />
            </div>

            <div>
              <label className="block font-semibold text-sm mb-1 text-gray-700">
                CSO Actual Signature <span className="text-red-500">*</span>
              </label>
              <input
                type="text"
                value={checklist.kycWatchlist.csoSignature}
                onChange={(e) => handleKycChange('csoSignature', e.target.value)}
                placeholder="Draw or type signature"
                className="w-full px-3 py-2 border border-gray-300 rounded focus:outline-none focus:ring-2 focus:ring-[#003366] focus:border-transparent text-sm"
                required
              />
            </div>

            <div>
              <label className="block font-semibold text-sm mb-1 text-gray-700">
                Date <span className="text-red-500">*</span>
              </label>
              <input
                type="date"
                value={checklist.kycWatchlist.csoDate}
                onChange={(e) => handleKycChange('csoDate', e.target.value)}
                className="w-full px-3 py-2 border border-gray-300 rounded focus:outline-none focus:ring-2 focus:ring-[#003366] focus:border-transparent text-sm"
                required
              />
            </div>
          </div>
        </div>

        {/* Approvals Section */}
        <div className="bg-white p-4 md:p-6 rounded-lg shadow-md mb-6">
          <h2 className="text-lg md:text-xl font-bold text-[#003366] border-b-2 border-[#003366] pb-2 mb-4">
            APPROVALS
          </h2>
          
          <div className="overflow-x-auto">
            <table className="w-full border-collapse text-xs md:text-sm">
              <thead>
                <tr className="bg-[#003366] text-white">
                  <th className="px-2 md:px-4 py-2 md:py-3 text-left font-semibold">APPROVING OFFICER</th>
                  <th className="px-2 md:px-4 py-2 md:py-3 text-left font-semibold">NAME</th>
                  <th className="px-2 md:px-4 py-2 md:py-3 text-left font-semibold">SIGNATURE</th>
                  <th className="px-2 md:px-4 py-2 md:py-3 text-left font-semibold">DATE</th>
                </tr>
              </thead>
              <tbody>
                {[
                  { key: 'customerServiceOfficer', label: 'Customer Service Officer' },
                  { key: 'headOfOperations', label: 'Head of Operations' },
                  { key: 'branchHead', label: 'Branch Head' },
                  { key: 'groupZonalHead', label: 'Group/Zonal Head (where applicable)' },
                ].map(({ key, label }) => (
                  <tr key={key} className="border-b border-gray-200 hover:bg-gray-50">
                    <td className="px-2 md:px-4 py-2 md:py-3 font-semibold text-gray-700">{label}</td>
                    <td className="px-2 md:px-4 py-2 md:py-3">
                      <input
                        type="text"
                        value={checklist.approvals[key].name}
                        onChange={(e) => handleApprovalChange(key, 'name', e.target.value)}
                        placeholder="Enter name"
                        className="w-full px-2 py-1 border border-gray-300 rounded focus:outline-none focus:ring-2 focus:ring-[#003366] focus:border-transparent text-sm"
                      />
                    </td>
                    <td className="px-2 md:px-4 py-2 md:py-3">
                      <input
                        type="text"
                        value={checklist.approvals[key].textSignature}
                        onChange={(e) => handleApprovalChange(key, 'textSignature', e.target.value)}
                        placeholder="Type signature"
                        className="w-full px-2 py-1 border border-gray-300 rounded focus:outline-none focus:ring-2 focus:ring-[#003366] focus:border-transparent text-sm font-cursive"
                        style={{ fontFamily: "'Segoe Script', 'Brush Script MT', cursive" }}
                      />
                    </td>
                    <td className="px-2 md:px-4 py-2 md:py-3">
                      <input
                        type="date"
                        value={checklist.approvals[key].date}
                        onChange={(e) => handleApprovalChange(key, 'date', e.target.value)}
                        className="w-full px-2 py-1 border border-gray-300 rounded focus:outline-none focus:ring-2 focus:ring-[#003366] focus:border-transparent text-sm"
                      />
                    </td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        </div>

        {/* Submit */}
        <div className="text-center mt-6">
          <button
            type="submit"
            className="w-full sm:w-auto px-8 sm:px-12 py-3 sm:py-4 bg-[#003366] text-white font-bold text-base sm:text-lg rounded-lg hover:bg-[#002244] transition-all duration-300 hover:shadow-lg hover:-translate-y-0.5"
          >
            Submit Checklist
          </button>
        </div>
      </form>
    </div>
  );
};

export default BankUseRequirementChecklist;
